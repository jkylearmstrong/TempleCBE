test_that("purl works on .qmd and .Rmd files", {
  template_dirs <- c(
    system.file("templates", package = "TempleCBE"),
    system.file("rmarkdown", "templates", package = "TempleCBE")
  )
  template_dirs <- template_dirs[nzchar(template_dirs)]
  files <- list.files(
    template_dirs,
    recursive = TRUE,
    pattern = "\\.(qmd|Rmd)$",
    full.names = TRUE
  )
  if (length(files) == 0) {
    files <- list.files(
      c("inst/templates", "inst/rmarkdown/templates", "../../inst/templates", "../../inst/rmarkdown/templates"),
      recursive = TRUE,
      pattern = "\\.(qmd|Rmd)$",
      full.names = TRUE
    )
  }
  expect_gt(length(files), 0)
  expect_true(any(grepl("\\.qmd$", files)))
  expect_true(any(grepl("\\.Rmd$", files)))
  # Isolate file output in a temporary directory so purled .R files do not pollute the repo
  withr::local_dir(withr::local_tempdir())
  res <- purrr::map_dfr(files, \(x) {
    out_file <- knitr::purl(x, quiet = TRUE)
    tibble::tibble(
      input = x,
      code  = readLines(out_file)
    )
  })
  expect_s3_class(res, "tbl_df")
  expect_gt(nrow(res), 0)
  expect_true(all(files %in% unique(res$input)))
  expect_true(any(grepl("\\.qmd$", res$input)))
  expect_true(any(grepl("\\.Rmd$", res$input)))
})
